cmake_minimum_required(VERSION 3.26)
project(DagFlowCapabilityCheck VERSION 1.0 LANGUAGES CXX)
include(CTest)

get_filename_component(DAGFLOW_SOURCE_DIR "../.." ABSOLUTE)
include("${DAGFLOW_SOURCE_DIR}/cmake/DagFlow.cmake")
option(CHECK_CUDA "Compile a CUDA target after configuring the capability" OFF)
option(CHECK_PROPERTY "Build and run a RapidCheck test against DagFlow" OFF)
set(DAGFLOW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(DAGFLOW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(DAGFLOW_ALLOCATOR system CACHE STRING "" FORCE)
set(DAGFLOW_COMPILER_CACHE none CACHE STRING "" FORCE)
set(DAGFLOW_FUZZ_SANITIZER undefined CACHE STRING "" FORCE)

set(_capabilities testing fuzzing property-testing build-info diagnostics)
if(CHECK_CUDA)
  list(APPEND _capabilities cuda)
endif()
dagflow_project(CAPABILITIES ${_capabilities})
add_subdirectory("${DAGFLOW_SOURCE_DIR}" dagflow)
dagflow_add_fuzzer(capability_fuzz SOURCES fuzz.cpp LIBRARIES DagFlow::DagFlow)
dagflow_add_test(capability_metadata SOURCES metadata.cpp LIBRARIES dagflow::build_info)
if(CHECK_PROPERTY)
  dagflow_add_property_test(capability_property SOURCES property.cpp LIBRARIES DagFlow::DagFlow)
endif()
if(CHECK_CUDA)
  add_library(capability_cuda STATIC probe.cu)
  add_dependencies(dagflow_tests capability_cuda)
endif()
dagflow_finalize_project()
